02 / EIP-712 schema

Non-executable by construction.

ElenchiArena.sol has no swap, router, mint or allowance-spending functions. Every signed action commits to keccak256("ELENCHI_EVIDENCE_ONLY_NON_EXECUTABLE_V1"). 

EIP712Domain(
  string  name,              // "ElenchiArena"
  string  version,           // "1"
  uint256 chainId,           // 46630 testnet | 4663 mainnet
  address verifyingContract
)

SimulatedAction(
  bytes32 runId,
  bytes32 policyHash,
  bytes32 scenarioHash,
  uint256 nonce,
  uint64  deadline,
  address token,
  address recipient,
  uint256 amount,
  uint16  maxSlippageBps,
  uint8   actionCategory,    // 0 SKIP · 1 EVIDENCE_SIMULATED_SWAP
  bytes32 evidenceNoticeHash
)
